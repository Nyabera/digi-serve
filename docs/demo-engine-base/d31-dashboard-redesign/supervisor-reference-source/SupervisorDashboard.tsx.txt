function Card({ children, className = "" }: { children: ReactNode; className?: string }) {
  return <section className={`sup-card ${className}`}>{children}</section>;
}

function ToneIcon({ icon: Icon, tone = "blue", small = false }: { icon: LucideIcon; tone?: Tone; small?: boolean }) {
  return <span className={`sup-tone-icon sup-tone-icon--${tone}${small ? " sup-tone-icon--small" : ""}`} aria-hidden="true"><Icon strokeWidth={2} /></span>;
}

function TextAction({ children, onClick, className = "" }: { children: ReactNode; onClick?: () => void; className?: string }) {
  return <button className={`sup-text-action ${className}`} type="button" onClick={onClick}>{children}</button>;
}

function PanelHeader({ title, meta, action, onAction }: { title: string; meta?: ReactNode; action?: string; onAction?: () => void }) {
  return (
    <div className="sup-panel-header">

export default function SupervisorDashboard({ data = supervisorDashboardData, embedded = true }: SupervisorDashboardProps) {
  const [toast, setToast] = useState("");
  const notify = (message: string) => { setToast(message); window.setTimeout(() => setToast(""), 2200); };
  const body = useMemo(() => <><DashboardHeader data={data} showTools={!embedded} notify={notify} /><DashboardBody data={data} notify={notify} /></>, [data, embedded]);

  return (
    <main className={`supervisor-dashboard supervisor-dashboard--${embedded ? "embedded" : "reference"}`} data-dashboard-version="supervisor-reference-v1">
      {embedded ? <div className="sup-dashboard-frame">{body}</div> : <div className="sup-reference-shell"><ReferenceSidebar notify={notify} /><div className="sup-reference-main">{body}</div></div>}
      <div className={`sup-toast${toast ? " is-visible" : ""}`} role="status" aria-live="polite"><Check />{toast}</div>
    </main>
  );
}
